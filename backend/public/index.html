<script>

  const DB_URL = " https://trailsafe-dineth988-default-rtdb.asia-southeast1.firebasedatabase.app/";   

  const POLL_MS = 5000;

  const STALE_MS = 120000;


  const token = new URLSearchParams(location.search).get("s");

  const who = document.getElementById("who");

  const badge = document.getElementById("badge");

  const info = document.getElementById("info");


  const map = L.map("map").setView([7.8731, 80.7718], 7);

  L.tileLayer("https://tile.openstreetmap.org/{z}/{x}/{y}.png", {

    maxZoom: 19, attribution: "&copy; OpenStreetMap contributors"

  }).addTo(map);

  let dot = null, halo = null, centred = false;


  function setBadge(text, cls) { badge.textContent = text; badge.className = cls || ""; }


  function render(lines, links, sos) {

    info.className = sos ? "sos" : "";

    info.replaceChildren();

    for (const line of lines) {

      const div = document.createElement("div");

      div.textContent = line;

      info.appendChild(div);

    }

    if (links && links.length) {

      const div = document.createElement("div");

      for (const l of links) {

        const a = document.createElement("a");

        a.textContent = l.label; a.href = l.href; a.target = "_blank"; a.rel = "noopener";

        div.appendChild(a);

      }

      info.appendChild(div);

    }

  }


  function ago(ms) {

    const s = Math.max(0, Math.round((Date.now() - ms) / 1000));

    if (s < 60) return s + " seconds ago";

    const m = Math.round(s / 60);

    return m < 60 ? m + " min ago" : Math.round(m / 60) + " h ago";

  }


  function show(d) {

    const name = String(d.ownerName || "Someone");

    who.textContent = name;


    if (Date.now() > Number(d.expiresAt)) {

      setBadge("Expired");

      render(["This Live Share has expired."]);

      return;

    }

    const sos = d.status === "sos";

    const stale = Date.now() - Number(d.updatedAt) > STALE_MS;

    setBadge(sos ? "SOS" : (stale ? "No recent update" : "Live"), sos ? "sos" : (stale ? "" : "live"));


    const lines = ["Last update: " + ago(Number(d.updatedAt))];

    if (sos && d.sosAt) lines.unshift(name + " triggered an SOS " + ago(Number(d.sosAt)) + ".");

    if (stale) lines.push("The phone may be offline or out of signal. The last known position is shown.");

    if (d.trailActive && typeof d.distanceMeters === "number")

      lines.push("Trail distance: " + (d.distanceMeters / 1000).toFixed(2) + " km");

    lines.push("Sharing ends at " + new Date(Number(d.expiresAt)).toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" }));


    const links = [];

    if (typeof d.lat === "number" && typeof d.lon === "number") {

      const lat = Number(d.lat), lon = Number(d.lon);

      const colour = sos ? "#B3261E" : "#2D5A3D";

      if (!dot) {

        halo = L.circle([lat, lon], { radius: d.accuracy || 20, color: colour, weight: 1, fillOpacity: 0.1 }).addTo(map);

        dot = L.circleMarker([lat, lon], { radius: 9, color: "#fff", weight: 3, fillColor: colour, fillOpacity: 1 }).addTo(map);

      } else {

        dot.setLatLng([lat, lon]).setStyle({ fillColor: colour });

        halo.setLatLng([lat, lon]).setRadius(d.accuracy || 20).setStyle({ color: colour });

      }

      if (!centred) { map.setView([lat, lon], 16); centred = true; }

      else if (!map.getBounds().contains([lat, lon])) map.panTo([lat, lon]);

      links.push({ label: "Open in Apple Maps", href: "https://maps.apple.com/?ll=" + lat + "," + lon });

      links.push({ label: "Open in Google Maps", href: "https://www.google.com/maps?q=" + lat + "," + lon });

    } else {

      lines.push("Waiting for the first location…");

    }

    render(lines, links, sos);

  }


  async function poll() {

    if (!token || token.length < 24) {

      setBadge("Invalid link"); render(["This link isn't valid."]); return;

    }

    try {

      const res = await fetch(DB_URL + "/shares/" + encodeURIComponent(token) + ".json", { cache: "no-store" });

      if (!res.ok) throw new Error("HTTP " + res.status);

      const data = await res.json();

      if (!data) { who.textContent = "TrailSafe Live Share"; setBadge("Ended"); render(["This Live Share has ended."]); }

      else show(data);

    } catch (e) {

      setBadge("Reconnecting…");

    }

    setTimeout(poll, POLL_MS);

  }

  poll();

</script>
